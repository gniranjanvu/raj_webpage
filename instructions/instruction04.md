```yaml
Fields:
  Basic:
    - title: string (required)
    - slug: string (auto-generated, editable)
    - category_id: select (required)
    - hero_image: file/URL (required)
    - short_description: string (required, max 200)
    - is_featured: boolean
    - display_order: number

  Content:
    - full_description: rich text (required)
    - tech_stack: tags array

  Media:
    - gallery_urls: file/URL array (reorderable)
    - video_urls: URL array (YouTube, Vimeo)

  Links:
    - action_buttons: array of {label, url, icon}

  Contributors:
    - contributors: array of {name, role, photo, profile_url}

  Engagement:
    - enable_comments: boolean
    - enable_likes: boolean
    - require_comment_approval: boolean

  SEO:
    - meta_title: string (optional)
    - meta_description: string (optional)
    - keywords: tags array

  Settings:
    - status: enum (draft, published, archived)
    - development_date: date
```
